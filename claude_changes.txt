          const deltaY = e.clientY - prevPointerY;
          targetRotY += deltaX * 0.008;
          targetRotX += deltaY * 0.008;
          prevPointerX = e.clientX;
          prevPointerY = e.clientY;

  loginTrigger.addEventListener('click', function(){ openModal(loginOverlay); refreshSession(); });
  function openAuth(mode){ setMsg(''); showStep(mode); openModal(loginOverlay); refreshSession(); }

  $('navFace').addEventListener('click', function(){ openAuth('signup'); });
  $('navSignup').addEventListener('click', function(){ openAuth('signup'); });
  $('mobLogin').addEventListener('click', function(){ openAuth('signin'); });
  $('mobSignup').addEventListener('click', function(){ openAuth('signup'); });
  $('loginCancel').addEventListener('click', function(){ resetLogin(); closeLogin(); });
  $('authClose').addEventListener('click', closeLogin);
  loginOverlay.addEventListener('click', function(e){ if(e.target === loginOverlay) closeLogin(); });
  document.querySelectorAll('.auth-tab').forEach(function(t){
    t.addEventListener('click', function(){ setMsg(''); showStep(t.dataset.mode); });
  });
  $('forgotBtn').addEventListener('click', function(){ setMsg(''); showStep('forgot'); });
  $('backBtn').addEventListener('click', function(){ setMsg(''); showStep('signin'); });

    if(sb){ try{ await sb.auth.signOut(); }catch(e){} }
    resetLogin();
    await refreshSession();

  // Links from emails: the page is opened with Supabase data in the address.
  function attachAuth(){
    sb.auth.onAuthStateChange(function(event){
      if(event === 'PASSWORD_RECOVERY'){
        cleanUrl();
        openModal(loginOverlay);
        showStep('newpw');
        setMsg('Choose a new password for your account.');
      } else if(event === 'SIGNED_IN' && isAuthCallback(window.location.hash)){
        cleanUrl();
        openModal(loginOverlay);
        $('signedInMsg').textContent = 'Your email is verified and you are signed in.';
        refreshSession();
      } else if(event === 'SIGNED_IN' || event === 'SIGNED_OUT'){
        refreshSession();
      }
    });
    refreshSession();
  }
  loadSupabase(0);
  (function showLinkErrors(){
    const hash = window.location.hash || '';
    if(!isAuthCallback(hash)) return;
    const params = new URLSearchParams(hash.replace(/^#/, ''));
    if(params.get('error') || params.get('error_code')){
      cleanUrl();
      openModal(loginOverlay);
      showStep('signin');
      setMsg('That link has expired or is not valid. Please request a new one.', true);
    }
  })();

  $('authForm').addEventListener('submit', async function(e){
    e.preventDefault();
    if(!sb){ setMsg('Login could not load. Please refresh the page and try again. If it still fails, a browser or security extension may be blocking it, or email CidexInfotech1@gmail.com.', true); return; }
    const email = $('authEmail').value.trim();
    const pw = pwInput.value;
    const newpw = $('authNewPw').value;
    setBusy(true);
    try {
      if(step === 'signin'){
        if(!email || !pw){ setMsg('Enter your email and password.', true); return; }
        const r = await sb.auth.signInWithPassword({ email: email, password: pw });
        if(r.error){
          if(/not confirmed/i.test(r.error.message)){
            await sb.auth.resend({ type: 'signup', email: email, options: { emailRedirectTo: SITE_URL } });
            setMsg('Your email is not verified yet. We sent you a new verification link.');
          } else {
            setMsg('Email or password is not right.', true);
          }
          return;
        }
        resetLogin(); closeLogin(); await refreshSession();
      }
      else if(step === 'signup'){
        if(!email || pw.length < 8){ setMsg('Use a valid email and a password of at least 8 characters.', true); return; }
        if(!$('authConsent').checked){ setMsg('Please accept the Privacy Policy and Terms & Conditions to create an account.', true); return; }
        const r = await sb.auth.signUp({ email: email, password: pw, options: { emailRedirectTo: SITE_URL, data: { terms_accepted_at: new Date().toISOString() } } });
        if(r.error){
          setMsg(/already/i.test(r.error.message) ? 'An account with this email already exists. Try signing in.' : 'We could not create the account. Please try again.', true);
          return;
        }
        pwInput.value = '';
        setMsg('Check your email. We sent a verification link to ' + email + '. Open it to activate your account.');
      }
      else if(step === 'forgot'){
        if(!email){ setMsg('Enter your email.', true); return; }
        await sb.auth.resetPasswordForEmail(email, { redirectTo: SITE_URL });
        setMsg('If an account exists for this email, we sent a password reset link. Open it to choose a new password.');
      }
      else if(step === 'newpw'){
        if(newpw.length < 8){ setMsg('Use at least 8 characters.', true); return; }
        const r = await sb.auth.updateUser({ password: newpw });
        if(r.error){ setMsg('We could not save the password. The link may have expired. Request a new one.', true); return; }
        resetLogin(); closeLogin(); await refreshSession();
      }
    } catch(err){
      setMsg('Something went wrong. Please try again.', true);
    } finally {
      setBusy(false);
    }
  });

  pwToggle.addEventListener('click', function(){
    const show = pwInput.type === 'password';
    pwInput.type = show ? 'text' : 'password';
    pwToggle.textContent = show ? 'Hide' : 'Show';
    pwToggle.setAttribute('aria-label', show ? 'Hide password' : 'Show password');
  });

  refreshSession();

  /* ---------- UTM TRACKING ---------- */
  const params = new URLSearchParams(window.location.search);
  const utmKeys = ['utm_source','utm_medium','utm_campaign','utm_term','utm_content'];
  let utm = {};
  const hasNewUtm = utmKeys.some(function(k){ return params.has(k); });
  if(hasNewUtm){
    utmKeys.forEach(function(k){ if(params.has(k)) utm[k] = params.get(k); });
    sessionStorage.setItem('cidex_utm', JSON.stringify(utm));
  } else {
    try{ utm = JSON.parse(sessionStorage.getItem('cidex_utm') || '{}'); }catch(e){ utm = {}; }
  }
  if(utm.utm_source){
    const tag = encodeURIComponent('Referred via: ' + utm.utm_source + (utm.utm_campaign ? ' / ' + utm.utm_campaign : ''));
    ['footerEmailLink'].forEach(function(id){
      const link = document.getElementById(id);
      if(link) link.href = 'mailto:CidexInfotech1@gmail.com?body=' + tag;
    });
  }

  /* ---------- NEWSLETTER FORM (with basic spam/bot protection) ---------- */
  const newsletterForm = document.getElementById('newsletterForm');
  const newsletterMsg = document.getElementById('newsletterMsg');
  const newsletterLoadedAt = Date.now();
  newsletterForm.addEventListener('submit', function(e){
    e.preventDefault();
    const email = document.getElementById('newsletterEmail').value.trim();
    const honeypot = document.getElementById('newsletterHoneypot').value;
    newsletterMsg.classList.remove('success','error');

    // Honeypot: a hidden field real visitors never fill in — bots that auto-fill every field trip it.
    if(honeypot){
      newsletterMsg.textContent = 'Subscribed. Pricing updates will land in your inbox.';
      newsletterMsg.classList.add('success');
      newsletterForm.reset();
      return; // silently drop — don't tip off the bot
    }
    // Time trap: a real person takes at least a couple seconds to read and type.
    if(Date.now() - newsletterLoadedAt < 2000){
      newsletterMsg.textContent = 'Please try that again.';
      newsletterMsg.classList.add('error');
      return;
    }
    if(!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)){
      newsletterMsg.textContent = 'Enter a valid email to subscribe.';
      newsletterMsg.classList.add('error');
      return;
    }
    newsletterMsg.textContent = 'Sign-ups aren\'t connected yet, so your email was not saved. Email CidexInfotech1@gmail.com to join the list.';
    newsletterMsg.classList.add('error');
    newsletterForm.reset();
  });


  /* ---------- LAST UPDATED + COPYRIGHT YEAR ---------- */
  const now = new Date();
  const lastMod = new Date(document.lastModified);
  const validDate = !isNaN(lastMod.getTime()) && lastMod.getFullYear() > 1990;
  const displayDate = validDate ? lastMod : now;
  document.querySelectorAll('.js-last-updated').forEach(function(el){
    if(el.getAttribute('data-mode') === 'year'){ el.textContent = now.getFullYear(); }
    else { el.textContent = displayDate.toLocaleDateString(undefined, { year:'numeric', month:'short', day:'numeric' }); }
  });
})();

/* ---------- AI PRICE ASSISTANT (talks to /api/chat) ---------- */
(function(){
  const fab = document.getElementById('aiFab');
  const panel = document.getElementById('aiPanel');
  const closeBtn = document.getElementById('aiClose');
  const form = document.getElementById('aiForm');
  const input = document.getElementById('aiInput');
  const sendBtn = document.getElementById('aiSend');
  const log = document.getElementById('aiLog');
  if(!fab || !panel || !form) return;

  let state = { service: null, answers: {} };
  let history = [];
  let estimates = [];      // every estimate card shown, so they can re-render when currency changes
  let busy = false;

  function currency(){
    try { return localStorage.getItem('cidex_currency') === 'INR' ? 'INR' : 'USD'; }
    catch(e){ return 'USD'; }
  }

  function addText(role, text){
    const d = document.createElement('div');
    d.className = 'ai-msg ' + role;
    d.textContent = text;
    log.appendChild(d);
    log.scrollTop = log.scrollHeight;
    return d;
  }

  function line(parent, left, right){
    const row = document.createElement('div');
    row.className = 'ai-line';
    const a = document.createElement('span'); a.textContent = left;
    const b = document.createElement('span'); b.textContent = right;
    row.appendChild(a); row.appendChild(b);
    parent.appendChild(row);
  }

  function renderCard(est, holder){
    const cur = currency();
    holder.textContent = '';
    const card = document.createElement('div');
    card.className = 'ai-card';

    const title = document.createElement('div');
    title.className = 'ai-card-title';
    title.textContent = est.label || 'Your estimate';
    card.appendChild(title);

    if(est.status === 'team'){
      const why = document.createElement('p');
      why.style.margin = '8px 0';
      why.textContent = 'Your project needs a personal assessment. ' + (est.reason || '') + ' Our team will review your requirements and send a suitable quotation.';
      card.appendChild(why);
      const c = document.createElement('div');
      c.textContent = 'Phone: ' + est.contact.phone;
      card.appendChild(c);
      const mail = document.createElement('a');
      mail.href = 'mailto:' + est.contact.email;
      mail.textContent = est.contact.email;
      card.appendChild(mail);
      holder.appendChild(card);
      return;
    }

    const total = document.createElement('div');
    total.className = 'ai-total';
    total.textContent = est.total[cur];
    card.appendChild(total);

    const period = document.createElement('div');
    period.className = 'ai-period';
    const periodText = est.period === 'monthly' ? 'per month' : 'one-time';
    period.textContent = (est.status === 'from' ? 'Starting price, ' : 'Estimate, ') + periodText + ' · excludes GST';
    card.appendChild(period);

    est.lines.forEach(function(l){ line(card, l.label, l[cur]); });

    const notes = [est.rounding_note, cur === 'USD' ? est.usd_note : '', est.from_note].filter(Boolean).join(' ');
    const note = document.createElement('div');
    note.className = 'ai-note';
    note.textContent = est.disclaimer + (notes ? ' ' + notes : '') + ' Management confirms final GST and any contract terms.';
    card.appendChild(note);

    holder.appendChild(card);
  }

  function addEstimate(est){
    const holder = document.createElement('div');
    holder.style.alignSelf = 'stretch';
    log.appendChild(holder);
    estimates.push({ est: est, holder: holder });
    renderCard(est, holder);
    log.scrollTop = log.scrollHeight;
  }

  function open(){
    panel.classList.add('open');
    fab.setAttribute('aria-expanded', 'true');
    if(!log.childElementCount){
      addText('bot', 'Hi. Tell me about your project and I will work out an estimate. You can switch between USD and INR at the top.');
    }
    input.focus();
  }
  function close(){
    panel.classList.remove('open');
    fab.setAttribute('aria-expanded', 'false');
    fab.focus();
  }

  fab.addEventListener('click', function(){ panel.classList.contains('open') ? close() : open(); });
  closeBtn.addEventListener('click', close);
  document.addEventListener('keydown', function(e){ if(e.key === 'Escape' && panel.classList.contains('open')) close(); });

  // Currency buttons inside the panel are handled by the main site script; re-draw cards after it runs.
  panel.querySelectorAll('.currency-toggle button').forEach(function(b){
    b.addEventListener('click', function(){
      setTimeout(function(){ estimates.forEach(function(e){ renderCard(e.est, e.holder); }); }, 0);
    });
  });

  form.addEventListener('submit', async function(e){
    e.preventDefault();
    const text = input.value.trim();
    if(!text || busy) return;
    input.value = '';
    addText('user', text);
    history.push({ role: 'user', content: text });
    busy = true;
    sendBtn.disabled = true;
    const waiting = addText('bot', '...');
    try {
      const res = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ messages: history.slice(-20), state: state, currency: currency() })
      });
      const data = await res.json();
      waiting.remove();
      if(data.reply){
        addText('bot', data.reply);
        history.push({ role: 'assistant', content: data.reply });
      }
      if(data.state){ state = data.state; }
      if(data.estimate){ addEstimate(data.estimate); }
    } catch(err){
      waiting.remove();
      addText('bot', 'Connection problem. Please try again, or contact the team at +91 87773 24473 or cidexinfotech1@gmail.com.');
    } finally {
      busy = false;
      sendBtn.disabled = false;
      input.focus();
    }
  });
})();
</script>
